#include <MD_MAX72xx.h>
#include <SPI.h>
#include <WiFi.h>
#include "esp_wpa2.h"
#include <HTTPClient.h>

#define HARDWARE_TYPE MD_MAX72XX::FC16_HW
#define DATA_PIN 13
#define CLK_PIN 14
#define CS_PIN 12
#define MAX_DEVICES 4

// ADC input pins for 4 matrices
int adcPins[MAX_DEVICES] = {32, 33, 34, 35};

// Button / Touch module input
#define TOUCH_PIN 27

MD_MAX72XX mx(HARDWARE_TYPE, DATA_PIN, CLK_PIN, CS_PIN, MAX_DEVICES);

// Wi-Fi & Server Config
const char* ssid = "IIITB-Milan";
const char* username = "iiitb\\Aryan.Sharma12";
const char* password = "F0y91zEG";
const char* serverName = "https://sensorvault.live/api/iot/taskarrows";

// Arrow patterns (Upside Down)
byte arrow_NE[8] = {
  B11100000, B11000000, B10100000, B00010000,
  B00001000, B00000100, B00000010, B00000001
};

byte arrow_NW[8] = {
  B00000001, B00000010, B00000100, B00001000,
  B00010000, B10100000, B11000000, B11100000
};

byte arrow_SE[8] = {
  B00000111, B00000011, B00000101, B00001000,
  B00010000, B00100000, B01000000, B10000000
};

byte arrow_SW[8] = {
  B10000000, B01000000, B00100000, B00010000,
  B00001000, B00000101, B00000011, B00000111
};

// Tick
byte tick[8] = {
  B01100000,
  B00110000,
  B00011000,
  B00001100,
  B00000110,
  B00000011,
  B00000110,
  B00001100,
  
};

// Cross
byte cross[8] = {
  B10000001,
  B01000010, 
  B00100100, 
  B00011000,
  B00011000, 
  B00100100, 
  B01000010, 
  B10000001
};

// ===== Helper Functions =====

void connectEnterpriseWiFi() {
  WiFi.disconnect(true);
  WiFi.mode(WIFI_STA);

  esp_wifi_sta_wpa2_ent_set_identity((uint8_t *)username, strlen(username));
  esp_wifi_sta_wpa2_ent_set_username((uint8_t *)username, strlen(username));
  esp_wifi_sta_wpa2_ent_set_password((uint8_t *)password, strlen(password));
  esp_wifi_sta_wpa2_ent_enable();

  WiFi.begin(ssid);

  Serial.println("\n==============================================");
  Serial.print("🔄 Connecting to WPA2-Enterprise Network: ");
  Serial.println(ssid);
  Serial.println("==============================================");

  int retries = 0;
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.print(".");
    retries++;
    if (retries > 25) {
      Serial.println("\n❌ WiFi Connection Timeout!");
      return;
    }
  }

  Serial.println("\n✅ Connected Successfully!");
  Serial.print("🌐 IP Address: ");
  Serial.println(WiFi.localIP());
  Serial.println("==============================================\n");
}

void displayPattern(uint8_t device, byte *pattern) {
  for (uint8_t row = 0; row < 8; row++) {
    mx.setRow(device, row, pattern[row]);
  }
}

void showAll(byte *pattern) {
  for (int i = 0; i < MAX_DEVICES; i++) {
    displayPattern(i, pattern);
  }
}

// ===== Main POST Function =====

bool sendPostWithDirections(String directions) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverName);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("Authorization", "bearer MaiSensorDataHoonMujheAndarJaneDo");

    String payload = "{\"current\":\"" + directions + "\"}";
    Serial.println("\n----------------------------------------------");
    Serial.print("➡️ Sending POST to: ");
    Serial.println(serverName);
    Serial.print("📦 Payload: ");
    Serial.println(payload);

    int httpResponseCode = http.POST(payload);
    Serial.print("📡 HTTP Response Code: ");
    Serial.println(httpResponseCode);

    if (httpResponseCode > 0) {
      String response = http.getString();
      Serial.println("🟢 Server Response:");
      Serial.println(response);
      Serial.println("----------------------------------------------\n");

      bool isCorrect = response.indexOf("\"isCorrect\":true") != -1;

      http.end();
      return isCorrect;
    } else {
      Serial.print("❌ POST Failed. Error Code: ");
      Serial.println(httpResponseCode);
      Serial.println("----------------------------------------------\n");
      http.end();
      return false;
    }
  } else {
    Serial.println("⚠️ WiFi Disconnected. Cannot send POST.");
    return false;
  }
}

// ===== Setup =====

void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n\n🚀 Booting ESP32 - 4x8x8 Matrix Controller\n");

  mx.begin();
  mx.control(MD_MAX72XX::INTENSITY, 6);
  mx.control(MD_MAX72XX::SHUTDOWN, false);
  mx.clear();

  for (int i = 0; i < MAX_DEVICES; i++) {
    pinMode(adcPins[i], INPUT);
    analogSetPinAttenuation(adcPins[i], ADC_11db);
  }

  pinMode(TOUCH_PIN, INPUT);

  connectEnterpriseWiFi();

  Serial.println("✅ System Ready!");
  Serial.println("🖐️ Press the button to send POST request.\n");
  Serial.println("------------------------------------------------------------");
}

// ===== Loop =====

void loop() {
  String allDirections = "";

  Serial.println("\n🔍 ADC Readings & Directions:");
  for (int i = 0; i < MAX_DEVICES; i++) {
    int mV = analogReadMilliVolts(adcPins[i]);
    String direction;

    if (mV < 250) {
      displayPattern(i, arrow_SW);
      direction = "SW";
    } else if (mV < 500) {
      displayPattern(i, arrow_NW);
      direction = "NW";
    } else if (mV < 1500) {
      displayPattern(i, arrow_NE);
      direction = "NE";
    } else {
      displayPattern(i, arrow_SE);
      direction = "SE";
    }

    allDirections += direction.substring(0, 2); // e.g. NE, NW, SE, SW
    Serial.printf("📟 Matrix %d | Pin %d | Voltage: %d mV | Direction: %s\n",
                  i, adcPins[i], mV, direction.c_str());
  }

  Serial.print("📦 Current String to Send: ");
  Serial.println(allDirections);
  Serial.println("------------------------------------------------------------");

  // When button pressed
  if (digitalRead(TOUCH_PIN) == HIGH) {
    Serial.println("\n🖐️ Button Press Detected!");
    bool isCorrect = sendPostWithDirections(allDirections);

    if (isCorrect) {
      Serial.println("✅ Correct → Showing Tick ✔️");
      showAll(tick);
      delay(1500);
    } else {
      Serial.println("❌ Incorrect → Showing Cross ✖️");
      showAll(cross);
      delay(2000);
    }

    mx.clear();
    delay(700); // debounce
  }


}
