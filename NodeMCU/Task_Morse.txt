#include <LiquidCrystal.h>
#include <Keypad.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include "esp_wpa2.h"

// ===== LCD Setup =====
LiquidCrystal lcd(13, 33, 32, 27, 26, 25);

// ===== Keypad Setup =====
const byte ROWS = 4;
const byte COLS = 4;

char keys[ROWS][COLS] = {
  {'1', '2', '3', 'A'},
  {'4', '5', '6', 'B'},
  {'7', '8', '9', 'C'},
  {'*', '0', '#', 'D'}
};

byte rowPins[ROWS] = {4, 5, 18, 19};
byte colPins[COLS] = {21, 16, 23, 14};

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// ===== Wi-Fi & Server Config =====
const char* ssid = "IIITB-Milan";
const char* username = "iiitb\\Aryan.Sharma12";
const char* password = "Removed MilanPass";
const char* serverName = "https://sensorvault.live/api/iot/taskmorse";

// ===== Variables =====
String enteredPIN = "";
const int PIN_LENGTH = 3;

// ===== WPA2 Enterprise Wi-Fi Connect =====
void connectEnterpriseWiFi() {
  WiFi.disconnect(true);
  WiFi.mode(WIFI_STA);

  esp_wifi_sta_wpa2_ent_set_identity((uint8_t *)username, strlen(username));
  esp_wifi_sta_wpa2_ent_set_username((uint8_t *)username, strlen(username));
  esp_wifi_sta_wpa2_ent_set_password((uint8_t *)password, strlen(password));
  esp_wifi_sta_wpa2_ent_enable();

  WiFi.begin(ssid);

  lcd.clear();
  lcd.print("Connecting WiFi");

  int retries = 0;
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    lcd.setCursor(retries % 16, 1);
    lcd.print(".");
    retries++;
    if (retries > 30) {
      lcd.clear();
      lcd.print("WiFi Failed!");
      return;
    }
  }

  lcd.clear();
  lcd.print("WiFi Connected!");
  delay(800);
}

// ===== Send POST Request =====
bool sendPostWithPIN(String pin) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverName);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("Authorization", "bearer MaiSensorDataHoonMujheAndarJaneDo");

    String payload = "{\"current\":\"" + pin + "\"}";
    Serial.print("➡️ Sending payload: ");
    Serial.println(payload);

    int httpResponseCode = http.POST(payload);

    if (httpResponseCode > 0) {
      String response = http.getString();
      Serial.println("🟢 Response:");
      Serial.println(response);

      bool isCorrect = response.indexOf("\"isCorrect\":true") != -1;
      http.end();
      return isCorrect;
    } else {
      Serial.print("❌ POST Failed, Code: ");
      Serial.println(httpResponseCode);
      http.end();
      return false;
    }
  } else {
    Serial.println("⚠️ WiFi Disconnected!");
    return false;
  }
}

// ===== Setup =====
void setup() {
  Serial.begin(115200);
  lcd.begin(16, 2);
  lcd.print("Booting System...");
  delay(1000);

  connectEnterpriseWiFi();

  lcd.clear();
  lcd.print("A-Enter D-Del");
  lcd.setCursor(0, 1);
  lcd.print("Enter PIN: ");
}

// ===== Loop =====
void loop() {
  char key = keypad.getKey();

  if (key) {
    if (key >= '0' && key <= '9') {
      if (enteredPIN.length() < PIN_LENGTH) {
        enteredPIN += key;
        lcd.setCursor(11, 1);
        lcd.print(enteredPIN);
      }
    }

    else if (key == 'D') {  // Delete
      if (enteredPIN.length() > 0) {
        enteredPIN.remove(enteredPIN.length() - 1);
        lcd.setCursor(11, 1);
        lcd.print("   ");  // clear space for 3-digit pin
        lcd.setCursor(11, 1);
        lcd.print(enteredPIN);
      }
    }

    else if (key == 'A') {  // Submit
      lcd.clear();
      lcd.print("Sending...");
      Serial.print("Submitting PIN: ");
      Serial.println(enteredPIN);

      bool success = sendPostWithPIN(enteredPIN);

      lcd.clear();
      if (success) {
        lcd.print("Access Granted");
      } else {
        lcd.print("Access Denied");
      }

      delay(1500);
      lcd.clear();
      lcd.print("A-Enter D-Del");
      lcd.setCursor(0, 1);
      lcd.print("Enter PIN: ");
      enteredPIN = "";
    }
  }
}
